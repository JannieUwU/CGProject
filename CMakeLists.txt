cmake_minimum_required(VERSION 3.16)
project(CGProject LANGUAGES CXX)
include(CTest)
add_library(cg_ef INTERFACE)
target_include_directories(cg_ef INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(cg_ef INTERFACE cxx_std_17)
add_executable(ef_tests tests/ef_tests.cpp)
target_link_libraries(ef_tests PRIVATE cg_ef)
if(BUILD_TESTING)
  add_test(NAME ef_foundation COMMAND ef_tests)
endif()
